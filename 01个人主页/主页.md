dv.el("div", "> CURRENT.FOCUS", {cls: "hp-section-title"});

// ---- 左侧：赛博时钟 + 迷你日历 ----
const clockWidget = document.createElement("div");
clockWidget.className = "hp-clock-widget";

const _now = new Date();
const _year = _now.getFullYear();
const _month = _now.getMonth();
const _today = _now.getDate();
const _firstDay = new Date(_year, _month, 1).getDay();
const _totalDays = new Date(_year, _month + 1, 0).getDate();
const _monthLabel = _now.toLocaleDateString('zh-CN', {year:'numeric', month:'long'});
const _weekdays = ['SU','MO','TU','WE','TH','FR','SA'];

let calHTML = `<div class="hp-cal-header">${_monthLabel}</div><div class="hp-cal-grid">`;
_weekdays.forEach(d => { calHTML += `<div class="hp-cal-weekday">${d}</div>`; });
for (let i = 0; i < _firstDay; i++) calHTML += '<div class="hp-cal-empty"></div>';
for (let d = 1; d <= _totalDays; d++) {
  const cls = d === _today ? "hp-cal-day hp-cal-today" : "hp-cal-day";
  calHTML += `<div class="${cls}">${String(d).padStart(2,'0')}</div>`;
}
calHTML += '</div>';

clockWidget.innerHTML = `
  <div class="hp-clock-time">--:--<span class="hp-clock-seconds">00</span></div>
  <div class="hp-clock-date">loading...</div>
  <div class="hp-clock-sep"></div>
  <div class="hp-calendar">${calHTML}</div>
`;

function _updateClock() {
  const t = new Date();
  const h = String(t.getHours()).padStart(2,'0');
  const m = String(t.getMinutes()).padStart(2,'0');
  const s = String(t.getSeconds()).padStart(2,'0');
  const ce = clockWidget.querySelector('.hp-clock-time');
  const de = clockWidget.querySelector('.hp-clock-date');
  if (ce) ce.innerHTML = `${h}:${m}<span class="hp-clock-seconds">${s}</span>`;
  if (de) de.textContent = t.toLocaleDateString('zh-CN', {year:'numeric',month:'2-digit',day:'2-digit',weekday:'short'});
}
_updateClock();
const _clockTimer = window.setInterval(() => {
  if (!clockWidget.isConnected) { clearInterval(_clockTimer); return; }
  _updateClock();
}, 1000);

clockFocusRow.appendChild(clockWidget);

// ---- 右侧：聚焦任务（从 YAML [[链接]] 自动解析）----
// 根据文件所在顶层目录自动映射图标和分类名
const _iconMap = {"02":"◉","03":"◈","04":"◉","05":"▤","06":"◻","07":"◷"};
const _catMap  = {"02":"INBOX","03":"PROJECT","04":"DOMAIN","05":"RESOURCE","06":"ARCHIVE","07":"LOG"};

const focusItems = [];
for (const raw of (dv.current().focus ?? [])) {
  const lnk = raw.link || raw;
  if (!lnk || !lnk.path) continue;
  const pg = dv.page(lnk.path);
  if (!pg) continue;
  const fp = pg.file.path;
  const pre = fp.substring(0, 2);
  const cat = _catMap[pre] || "VAULT";
  const parentFolder = pg.file.folder.split("/").pop()
    .replace(/^\d+/, "").replace(/[()（）_]/g, " ").trim();
  focusItems.push({
    icon: _iconMap[pre] || "◇",
    name: lnk.display || pg.file.name,
    meta: parentFolder ? `${cat} // ${parentFolder}` : cat,
    status: String(raw.status || "pending"),
    path: fp
  });
}

const focusBox = document.createElement("div");
focusBox.className = "hp-focus";
const focusList = document.createElement("ul");
focusList.className = "hp-focus-list";

for (const item of focusItems) {
  const li = document.createElement("li");
  li.className = "hp-focus-item";
  const statusLabel = item.status === "active" ? "RUNNING" : "QUEUED";
  li.innerHTML = `
    <span class="hp-focus-icon">${item.icon}</span>
    <div class="hp-focus-info">
      <div class="hp-focus-name">${item.name}</div>
      <div class="hp-focus-meta">${item.meta}</div>
    </div>
    <span class="hp-focus-status ${item.status}">${statusLabel}</span>
  `;
  li.addEventListener("click", () => {
    app.workspace.openLinkText(item.path, "");
  });
  focusList.appendChild(li);
}
focusBox.appendChild(focusList);
clockFocusRow.appendChild(focusBox);

dv.el("hr", "", {cls: "hp-divider"});

    !p.file.path.startsWith(".") &&
    !p.file.name.startsWith("Pasted image") &&
    p.file.ext === "md"

  .sort(p => p.file.mtime, "desc")
  .limit(10);

  const li = document.createElement("li");

  const ms = Date.now() - new Date(page.file.mtime).getTime();

  if (ms < 60000) timeStr = "刚刚";
  else if (ms < 3600000) timeStr = Math.floor(ms / 60000) + " 分钟前";
  else if (ms < 86400000) timeStr = Math.floor(ms / 3600000) + " 小时前";
  else timeStr = Math.floor(ms / 86400000) + " 天前";

    .replace(/^\d+/, "")
    .replace(/[()（）]/g, "");
